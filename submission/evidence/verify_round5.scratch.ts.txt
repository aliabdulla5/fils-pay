import assert from "node:assert/strict";
import { createHmac, randomUUID } from "node:crypto";
import { Pool } from "pg";
import { buildApp } from "./apps/api/src/app.js";
import { loadConfig } from "./apps/api/src/config.js";
import { parseCreditRequestInput } from "./apps/api/src/admin/creditRequestValidation.js";
const pool = new Pool({ connectionString: process.env.DATABASE_URL });
const SECRET = "acqsec_verify";
const config = loadConfig({ ...process.env, NODE_ENV: "test", ACQUIRER_CALLBACK_SIGNING_SECRET: SECRET });
const app = buildApp(config, pool);
await app.ready();
// a real transaction to attach the note to
const m = (await pool.query(`INSERT INTO merchants (display_name, mdr_bps, status, test_mode) VALUES ($1, 250, 'active', false) RETURNING id`, [`cb-${randomUUID()}`])).rows[0].id;
const t = (await pool.query(`INSERT INTO transactions (merchant_id, amount_fils, status, acquirer_reference) VALUES ($1, 1000, 'authorized', $2) RETURNING id`, [m, `ref-${randomUUID()}`])).rows[0].id;
const raw = `{\n  "callbackId": "acqcb_${randomUUID()}",\n  "type": "fraud_review.completed",\n  "transactionId": "${t}",\n  "occurredAt": "2026-10-30T10:00:00Z",\n  "outcome": "cleared",\n  "note": "caf\\u00e9 ok"\n}`;
const ts = Math.floor(Date.now() / 1000);
const sig = createHmac("sha256", SECRET).update(`${ts}.${raw}`).digest("hex");
const res = await app.inject({ method: "POST", url: "/v1/acquirer/callbacks", headers: { "content-type": "application/json", "acquirer-callback-signature": `t=${ts},v1=${sig}` }, payload: raw });
assert.equal(res.statusCode, 200, res.body);
console.log("PASS callback signed over its raw (pretty-printed, escaped) bytes is accepted");
const tampered = await app.inject({ method: "POST", url: "/v1/acquirer/callbacks", headers: { "content-type": "application/json", "acquirer-callback-signature": `t=${ts},v1=${sig}` }, payload: raw.replace("cleared", "confirmed_fraud") });
assert.equal(tampered.statusCode, 401, tampered.body);
console.log("PASS tampered raw body still rejected");
const bad = await app.inject({ method: "POST", url: "/v1/acquirer/callbacks", headers: { "content-type": "application/json" }, payload: "{not json" });
assert.equal(bad.statusCode, 400, bad.body);
console.log("PASS malformed JSON -> 400");
const tomorrow = new Date(Date.now() + 3 * 3600_000 + 86_400_000).toISOString().slice(0, 10);
assert.throws(() => parseCreditRequestInput({ baseAmountFils: 1000, reasonCode: "goodwill", effectiveDate: tomorrow }), /today or earlier/);
parseCreditRequestInput({ baseAmountFils: 1000, reasonCode: "goodwill", effectiveDate: new Date().toISOString().slice(0, 10) });
console.log("PASS future credit effective date -> 400; today (UTC) accepted");
await app.close(); await pool.end();
