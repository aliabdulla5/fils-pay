import { randomUUID } from "node:crypto";
import { Pool } from "pg";
import { authorizeTransaction } from "./apps/api/src/payments/authorize.js";
import { captureTransaction } from "./apps/api/src/payments/capture.js";
import { applyCaptureDiscountAdjustment } from "./apps/api/src/payments/discountAdjustment.js";
import { createAcquirerSimulator } from "./apps/api/src/acquirer/simulator.js";
import { loadConfig } from "./apps/api/src/config.js";
import { toFils } from "@filspay/money";
const pool = new Pool({ connectionString: process.env.DATABASE_URL });
const acq = createAcquirerSimulator({ ...loadConfig({ ...process.env, NODE_ENV: "test", ACQUIRER_FAILURE_RATE: "0", ACQUIRER_LATENCY_MIN_MS: "0", ACQUIRER_LATENCY_MAX_MS: "0" }) });
const m = (await pool.query(`INSERT INTO merchants (display_name, mdr_bps, negotiated_discount_bps, status, test_mode, onboarding_status) VALUES ($1, 250, 1000, 'active', false, 'approved') RETURNING id`, [`disc-${randomUUID()}`])).rows[0].id;
const card = { pan: "4242424242424242", expiryMonth: 12, expiryYear: 2030, cvv: "123" };
const auth = await authorizeTransaction(pool, acq, { merchantId: m, amountFils: toFils(1_000_100), card: card as any, idempotencyKey: randomUUID(), actor: { type: "system" } as any });
const txnId = (auth.body as any).id;
const actor = { type: "system" as const, id: m };
const k1 = randomUUID();
const c1 = await captureTransaction(pool, acq, { transactionId: txnId, merchantId: m, amountFils: toFils(100), idempotencyKey: k1, actor });
await applyCaptureDiscountAdjustment(pool, { merchantId: m, transactionId: txnId });
const c2 = await captureTransaction(pool, acq, { transactionId: txnId, merchantId: m, amountFils: toFils(1_000_000), idempotencyKey: randomUUID(), actor });
await applyCaptureDiscountAdjustment(pool, { merchantId: m, transactionId: txnId });
const sum = async () => (await pool.query(`SELECT count(*)::int n, COALESCE(sum(amount_fils),0)::int s FROM ledger_entries WHERE transaction_id=$1 AND direction='credit' AND description='capture: negotiated discount adjustment'`, [txnId])).rows[0];
console.log("statuses", auth.statusCode, c1.statusCode, c2.statusCode, "after two captures:", await sum());
const replay = await captureTransaction(pool, acq, { transactionId: txnId, merchantId: m, amountFils: toFils(100), idempotencyKey: k1, actor });
if (replay.statusCode === 200) await applyCaptureDiscountAdjustment(pool, { merchantId: m, transactionId: txnId });
console.log("after replaying capture 1:", await sum());
await pool.end();
